WITH req (tabla, columna) AS (
  SELECT 'D_OFFERPALETTE', c FROM (
    SELECT 'EXTID$' c FROM dual UNION ALL SELECT 'CAMPAIGNID' FROM dual
    UNION ALL SELECT 'REASONCODE' FROM dual UNION ALL SELECT 'REASONDESC' FROM dual
    UNION ALL SELECT 'APPTSTARTDATE_TM' FROM dual UNION ALL SELECT 'APPTENDDATE_TM' FROM dual
    UNION ALL SELECT 'EXPIRYDATE_TM' FROM dual UNION ALL SELECT 'OFFERCREATIONDATE_TM' FROM dual
    UNION ALL SELECT 'AVAILABILITY' FROM dual
    UNION ALL SELECT 'USERFIELD_' || LEVEL FROM dual CONNECT BY LEVEL <= 100)
  UNION ALL SELECT 'D_CAMPAIGN', 'CMPID' FROM dual
  UNION ALL SELECT 'D_CAMPAIGN', 'PRDCODE' FROM dual
  UNION ALL SELECT 'D_CAMPAIGN', 'PRDTYPE' FROM dual
  UNION ALL SELECT 'D_CAMPAIGN', 'FULFILLMENTPROCID' FROM dual
  UNION ALL SELECT 'D_PRODUCT', 'PRDCODE' FROM dual
  UNION ALL SELECT 'D_PRODUCT', 'DESCRIPTION' FROM dual
  UNION ALL SELECT 'D_PRODUCT', 'PRDFAMILY' FROM dual
  UNION ALL SELECT 'D_WAVECHINFO', 'EXTID$' FROM dual
  UNION ALL SELECT 'D_WAVECHINFO', 'CRPCATEGORYID' FROM dual
)
SELECT r.tabla, r.columna AS columna_que_falta
  FROM req r
 WHERE NOT EXISTS (SELECT 1 FROM all_tab_columns c
                    WHERE c.table_name = r.tabla AND c.column_name = r.columna)
 ORDER BY r.tabla, r.columna;



SELECT table_name, column_id, column_name
  FROM all_tab_columns
 WHERE table_name IN ('D_OFFERPALETTE', 'D_WAVECHINFO')
 ORDER BY table_name, column_id;
